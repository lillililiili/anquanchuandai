99:   Future<void> initialize({bool applyInitial = true}) async {
100:     final scope = _scopeGeneration;
101:     final restored = await _store.read();
102:     if (!_acceptScope(scope)) return;
103:     if (restored != null) {
104:       filters = restored.filters;
105:       current = restored.current;
106:       scrollOffset = restored.scrollOffset;
107:       _drafts
108:         ..clear()
109:         ..addAll(restored.drafts);
110:     }
111:     if (applyInitial) {
112:       filters = filters.copyWith(
113:         personId: initialPersonId ?? filters.personId,
114:         taskId: initialTaskId ?? filters.taskId,
115:         claimantUserId: initialClaimantUserId ?? filters.claimantUserId,
116:         escalated: initialEscalated ?? filters.escalated,
117:         status: initialStatus ?? filters.status,
118:         type: initialType ?? filters.type,
119:       );
120:     }
121:     final target = applyInitial
122:         ? initialEventId ?? restored?.selectedEventId ?? ''
123:         : restored?.selectedEventId ?? '';
124:     await reload(current: current);
125:     if (target.isNotEmpty && _acceptScope(scope)) await select(target);
126:   }

162:   Future<void> reload({int? current}) async {
163:     final request = ++_pageGeneration;
164:     final scope = _scopeGeneration;
165:     final targetPage = current ?? this.current;
166:     final requestedFilters = filters;
167:     loadingMore = false;
168:     loadMoreError = null;
169:     loading = true;
170:     errorMessage = null;
171:     notifyListeners();
172:     unawaited(_reloadInboxCount(scope));
173:     try {
174:       // Rebuild all previously loaded batches so refreshing/returning never
175:       // replaces a continuous list with only its final batch.
176:       final rows = <String, WearEvent>{};
177:       var reachedEnd = false;
178:       var loadedPage = 1;
179:       var latestTotal = 0;
180:       for (var number = 1; number <= targetPage; number++) {
181:         final page = await gateway.fetchPage(
182:           requestedFilters,
183:           number,
184:           pageSize,
185:         );
186:         if (!_acceptPage(scope, request)) return;
187:         for (final event in page.records) {
188:           rows[event.id] = event;
189:         }
190:         loadedPage = page.current;
191:         latestTotal = page.total;
192:         reachedEnd =
193:             page.records.isEmpty ||
194:             page.current < number ||
195:             page.current * pageSize >= page.total;
196:         if (reachedEnd) break;
197:       }
198:       records = rows.values.toList();
199:       total = latestTotal;
200:       this.current = loadedPage;
201:       _endReached = reachedEnd;
202:     } catch (error) {
203:       if (!_acceptPage(scope, request)) return;
204:       if (error is! EventStaleScope) errorMessage = error.toString();
205:     } finally {
206:       if (_acceptPage(scope, request)) {
207:         loading = false;
208:         notifyListeners();
209:         _persist();
210:       }
211:     }
212:   }

231:   Future<void> loadMore() async {
232:     if (loading || loadingMore || writing || !hasMore) return;
233:     final request = ++_pageGeneration;
234:     final scope = _scopeGeneration;
235:     final targetPage = current + 1;
236:     loadingMore = true;
237:     loadMoreError = null;
238:     notifyListeners();
239:     try {
240:       final page = await gateway.fetchPage(filters, targetPage, pageSize);
241:       if (!_acceptPage(scope, request)) return;
242:       final rows = {for (final event in records) event.id: event};
243:       for (final event in page.records) {
244:         rows[event.id] = event;
245:       }
246:       records = rows.values.toList();
247:       total = page.total;
248:       _endReached =
249:           page.records.isEmpty ||
250:           page.current < targetPage ||
251:           page.current * pageSize >= page.total;
252:       current = page.current;
253:     } catch (error) {
254:       if (_acceptPage(scope, request) && error is! EventStaleScope) {
255:         loadMoreError = '后续事件加载失败';
256:       }
257:     } finally {
258:       if (_acceptPage(scope, request)) {
259:         loadingMore = false;
260:         notifyListeners();
261:         unawaited(_persist());
262:       }
263:     }

357:   bool can(EventCommand command) {
358:     final event = selected;
359:     if (event == null || !EventPolicy.can(command, event, actor)) return false;
360:     if (command == EventCommand.ack &&
361:         (_ackedLocally.contains(event.id) ||
362:             (actor.userName.isNotEmpty &&
363:                 actions.any(
364:                   (action) =>
365:                       action.action == 'ack' && action.actor == actor.userName,
366:                 )))) {
367:       return false;
368:     }
369:     return true;
370:   }

372:   Future<bool> execute(EventCommand command) async {
373:     final event = selected;
374:     if (writing || event == null || !can(command)) {
375:       return false;
376:     }
377:     writing = true;
378:     errorMessage = null;
379:     conflictMessage = null;
380:     successMessage = null;
381:     notifyListeners();
382:     final scope = _scopeGeneration;
383:     final selectionGeneration = _detailGeneration;
384:     try {
385:       final latest = await gateway.execute(command, event, draftFor(event.id));
386:       if (!_acceptScope(scope)) return false;
387:       if (command != EventCommand.ack) _drafts.remove(event.id);
388:       if (command == EventCommand.ack) _ackedLocally.add(event.id);
389:       successMessage = _successText(command);
390:       if (selectionGeneration != _detailGeneration ||
391:           selected?.id != event.id) {
392:         await reload(current: current);
393:         return true;
394:       }
395:       selected = latest;
396:       try {
397:         await _refreshAfterWrite(event.id, scope);
398:       } on EventStaleScope {
399:         return true;
400:       } catch (error) {
401:         if (_acceptScope(scope)) {
402:           errorMessage = '操作已成功，但最新列表加载失败：$error';
403:         }
404:       }
405:       return true;
406:     } on EventConflict catch (conflict) {
407:       if (!_acceptScope(scope)) return false;
408:       conflictMessage = conflict.message;
409:       if (selectionGeneration != _detailGeneration ||
410:           selected?.id != event.id) {
411:         await reload(current: current);
412:         return false;
413:       }
414:       try {
415:         await _refreshAfterWrite(event.id, scope);
416:       } on EventStaleScope {
417:         return false;
418:       } catch (error) {
419:         if (_acceptScope(scope)) {
420:           errorMessage = '状态已冲突，重新加载失败：$error';
421:         }
422:       }
423:       return false;
424:     } catch (error) {
425:       if (_acceptScope(scope) && error is! EventStaleScope) {
426:         errorMessage = error.toString();
427:       }
428:       return false;
429:     } finally {
430:       if (_acceptScope(scope)) {
431:         writing = false;
432:         notifyListeners();
433:         _persist();
434:       }
435:     }
436:   }

438:   Future<void> refreshFromSignal() async {
439:     final request = ++_refreshGeneration;
440:     final scope = _scopeGeneration;
441:     final id = selected?.id;
442:     await reload(current: current);
443:     if (!_acceptScope(scope) || request != _refreshGeneration) return;
444:     if (id != null && id.isNotEmpty) await select(id);
445:   }

447:   Future<void> _refreshAfterWrite(String eventId, int scope) async {
448:     final detail = await Future.wait<Object>([
449:       gateway.fetchDetail(eventId),
450:       gateway.fetchActions(eventId),
451:     ]);
452:     if (!_acceptScope(scope)) return;
453:     selected = detail[0] as WearEvent;
454:     actions = detail[1] as List<EventAction>;
455:     await reload(current: current);
456:   }

458:   Future<void> _reloadInboxCount(int scope) async {
459:     final request = ++_countGeneration;
460:     try {
461:       final result = await gateway.fetchInboxCount();
462:       if (_acceptScope(scope) && request == _countGeneration) {
463:         inboxCount = result;
464:         notifyListeners();
465:       }
466:     } on EventStaleScope {
467:       // A replacement scope owns the next count.
468:     } catch (_) {
469:       // The authoritative page remains usable if this supplementary count fails.
470:     }
471:   }

473:   String _successText(EventCommand command) => switch (command) {
474:     EventCommand.ack => '已确认看见该事件',
475:     EventCommand.claim => '事件已认领',
476:     EventCommand.handle => '处置记录已提交',
477:     EventCommand.transfer => '事件已转交',
478:     EventCommand.close => '事件已关闭',
479:     EventCommand.reopen => '事件已重开',
480:     EventCommand.assignTask => '事件关联任务已确认',
481:   };

483:   bool _acceptScope(int scope) => !_disposed && scope == _scopeGeneration;
484:   bool _acceptPage(int scope, int request) =>
485:       _acceptScope(scope) && request == _pageGeneration;
486:   bool _acceptDetail(int scope, int request) =>
487:       _acceptScope(scope) && request == _detailGeneration;

489:   Future<void> _persist() {
490:     final store = _store;
491:     final value = EventWorkspaceState(
492:       filters: filters,
493:       current: current,
494:       selectedEventId: selected?.id ?? '',
495:       scrollOffset: scrollOffset,
496:       drafts: Map.unmodifiable(_drafts),
497:     );
498:     final operation = _persistChain.then((_) => store.write(value));
499:     _persistChain = operation.catchError((_) {});
500:     return operation;
501:   }

503:   @override
504:   void dispose() {

507:     _pageGeneration++;
508:     _detailGeneration++;
509:     _refreshGeneration++;
510:     _countGeneration++;
