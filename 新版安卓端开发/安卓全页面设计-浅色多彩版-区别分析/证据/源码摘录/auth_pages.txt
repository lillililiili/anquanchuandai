52:     var text = raw?.toString().trim() ?? '';

56:     text = text.replaceAll(RegExp(r'\s'), '');

495: class _LocalCaptchaView extends StatelessWidget {
496:   const _LocalCaptchaView({required this.code});
497: 
498:   final String code;
499: 
500:   @override
501:   Widget build(BuildContext context) {
502:     return ColoredBox(
503:       color: const Color(0xFFF4F7FF),
504:       child: CustomPaint(
505:         painter: _CaptchaPainter(code),
506:         child: const SizedBox.expand(),
507:       ),
508:     );
509:   }
510: }
511: 
512: class _CaptchaPainter extends CustomPainter {
513:   _CaptchaPainter(this.code);
514:   final String code;
515: 
516:   static const _colors = [
517:     Color(0xFF16A34A),
518:     Color(0xFFEA580C),
519:     Color(0xFF2563EB),
520:     Color(0xFF7C3AED),
521:   ];
522: 
523:   @override
524:   void paint(Canvas canvas, Size size) {
525:     final noise = Paint()
526:       ..color = const Color(0x332563EB)
527:       ..strokeWidth = 1;
528:     for (var i = 0; i < 6; i++) {
529:       canvas.drawLine(
530:         Offset(size.width * (0.05 + i * 0.15), 4),
531:         Offset(size.width * (0.2 + i * 0.12), size.height - 4),
532:         noise,
533:       );
534:     }
535:     for (var i = 0; i < code.length; i++) {
536:       final tp = TextPainter(
537:         text: TextSpan(
538:           text: code[i],
539:           style: TextStyle(
540:             color: _colors[i % _colors.length],
541:             fontSize: size.height * 0.62,
542:             fontWeight: FontWeight.w800,
543:             fontStyle: i.isOdd ? FontStyle.italic : FontStyle.normal,
544:           ),
545:         ),
546:         textDirection: TextDirection.ltr,
547:       )..layout();
548:       final dx = size.width * (0.08 + i * 0.22);
549:       final dy = (size.height - tp.height) / 2 + (i.isEven ? -2 : 3);
550:       canvas.save();
551:       canvas.translate(dx + tp.width / 2, dy + tp.height / 2);
552:       canvas.rotate((i - 1.5) * 0.12);
553:       tp.paint(canvas, Offset(-tp.width / 2, -tp.height / 2));
554:       canvas.restore();
555:     }
556:   }
557: 
558:   @override
559:   bool shouldRepaint(covariant _CaptchaPainter oldDelegate) =>
560:       oldDelegate.code != code;
561: }
