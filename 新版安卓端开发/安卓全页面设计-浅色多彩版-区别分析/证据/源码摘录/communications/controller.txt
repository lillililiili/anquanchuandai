95:   Future<void> startCall({
96:     bool video = false,
97:     String? eventId,
98:     String kind = 'single',
99:   }) async {
100:     if (_sessionOperationPending || busy || hasActiveOwnedSession) {
101:       statusMessage = '请先结束当前通话';
102:       _notify();
103:       return;
104:     }
105:     final device = selectedDevice;
106:     if (device == null) {
107:       statusMessage = '请先选择人员装备';
108:       _notify();
109:       return;
110:     }
111:     if (video ? !policy.canStartVideo(device) : !policy.canStartVoice(device)) {
112:       statusMessage = video ? '该设备不支持视频通话或当前账号无权限' : '该设备不支持通话或当前账号无权限';
113:       _notify();
114:       return;
115:     }
116: 
117:     final operation = _beginSessionOperation('正在创建通话…');
118:     try {
119:       final started = await gateway.startCall(
120:         deviceId: device.id,
121:         eventId: eventId,
122:         kind: kind,
123:         video: video,
124:         idempotencyKey: _idempotencyKey('call'),
125:       );
126:       if (!_acceptsOperation(operation)) return;
127:       _replaceActiveCall(started.session, fresh: true);
128:       _schedulePolling();
129:       if (started.session.demo || started.credentials?.demo == true) {
130:         rtcPhase = WearRtcPhase.demo;
131:         statusMessage = '演示会话已建立，未连接真实设备';
132:         return;
133:       }
134:       if (!policy.canJoin(started.session)) {
135:         statusMessage = policy.owns(started.session)
136:             ? started.session.statusLabel
137:             : '仅发起人可以加入该通话';
138:         return;
139:       }
140:       var credentials = started.credentials;
141:       if (credentials == null) {
142:         credentials = await gateway.credentialsFor(started.session.id);
143:         if (!_acceptsCallOperation(operation, started.session.id)) return;
144:       }
145:       await _join(operation, started.session, credentials);
146:       if (!_acceptsCallOperation(operation, started.session.id)) return;
147:     } on RtcOperationCancelled {
148:       if (_acceptsOperation(operation)) {
149:         rtcPhase = WearRtcPhase.idle;
150:         statusMessage = '通话加入已取消';
151:       }
152:     } on RtcPermissionDenied catch (error) {
153:       if (!_acceptsOperation(operation)) return;
154:       rtcPhase = WearRtcPhase.failed;
155:       statusMessage = error.video ? '请允许麦克风和摄像头权限后重试' : '请允许麦克风权限后重试';
156:     } catch (error) {
157:       if (!_acceptsOperation(operation)) return;
158:       rtcPhase = WearRtcPhase.failed;
159:       statusMessage = _errorMessage(error, '通话建立失败');
160:     } finally {
161:       _finishSessionOperation(operation);
162:     }
163:   }

220:   Future<void> _join(
221:     int operation,
222:     CallSession session,
223:     RtcCredentials credentials,
224:   ) async {
225:     if (credentials.demo) {
226:       rtcPhase = WearRtcPhase.demo;
227:       statusMessage = '演示会话已建立，未连接真实设备';
228:       return;
229:     }
230:     if (!credentials.isUsable || credentials.isExpired) {
231:       throw const RtcJoinFailure('通话凭证无效或已过期');
232:     }
233:     if (!_acceptsCallOperation(operation, session.id)) {
234:       throw const RtcOperationCancelled();
235:     }
236: 
237:     rtcPhase = WearRtcPhase.joining;
238:     statusMessage = '正在加入 RTC 频道…';
239:     final outcome = Completer<void>();
240:     _joinOutcome = outcome;
241:     _joinOutcomeOperation = operation;
242:     _notify();
243:     await rtc.join(
244:       credentials,
245:       RtcCallbacks(
246:         onLocalJoined: () =>
247:             unawaited(_markLocalJoined(operation, session.id, credentials.uid)),
248:         onRemoteJoined: (uid) => _onRemoteJoined(operation, session.id, uid),
249:         onRemoteLeft: (uid) => _onRemoteLeft(operation, session.id, uid),
250:         onTokenExpiring: () => unawaited(_refreshToken(operation, session.id)),
251:         onFailure: (message) =>
252:             _handleRtcFailure(operation, session.id, message),
253:       ),
254:     );
255:     if (!_acceptsCallOperation(operation, session.id)) {
256:       await rtc.leave();
257:       throw const RtcOperationCancelled();
258:     }
259:     _rtcActive = true;
260:     await outcome.future;
261:     if (!_acceptsCallOperation(operation, session.id)) {
262:       throw const RtcOperationCancelled();
263:     }
264:   }

346:   Future<void> refreshActiveCall() {
347:     final current = activeCall;
348:     if (_disposed || current == null || current.id.isEmpty) {
349:       return Future<void>.value();
350:     }
351:     final callGeneration = _callGeneration;
352:     final existing = _pollInFlight;
353:     if (existing != null &&
354:         _pollInFlightCallGeneration == callGeneration &&
355:         _pollInFlightCallId == current.id) {
356:       return existing;
357:     }
358:     final request = ++_pollRequestGeneration;
359:     late final Future<void> poll;
360:     poll = _refreshActiveCallOnce(current, callGeneration, request)
361:         .whenComplete(() {
362:           if (identical(_pollInFlight, poll)) {
363:             _pollInFlight = null;
364:             _pollInFlightCallGeneration = null;
365:             _pollInFlightCallId = null;
366:           }
367:         });
368:     _pollInFlight = poll;
369:     _pollInFlightCallGeneration = callGeneration;
370:     _pollInFlightCallId = current.id;
371:     return poll;
372:   }

374:   Future<void> _refreshActiveCallOnce(
375:     CallSession current,
376:     int callGeneration,
377:     int request,

379:     try {

381:       if (!_acceptsPoll(callGeneration, request, current.id)) return;

391:     } catch (error) {
392:       if (!_acceptsPoll(callGeneration, request, current.id)) return;
393:       _serverStateFresh = false;
394:       if (rtcPhase != WearRtcPhase.failed) {
395:         statusMessage = _errorMessage(error, '通话状态刷新失败');
396:       }
397:     }
398:     _notify();
399:   }

432:   Future<void> hangUp() async {
433:     final call = activeCall;
434:     if (_disposed || _ending || call == null || !policy.canEnd(call)) return;
435:     _ending = true;
436:     busy = true;
437:     _sessionOperationPending = true;

460:         _sessionOperationPending = false;
461:         _ending = false;
462:         busy = false;

484:   Future<void> sendTts({String? eventId}) async {
485:     final device = selectedDevice;
486:     final text = ttsText.trim();
487:     if (device == null || !policy.canSendTts(device)) {
488:       ttsMessage = '该设备不支持播报或当前账号无权限';
489:       _notify();
490:       return;
491:     }
492:     if (text.isEmpty) {
493:       ttsMessage = '请输入播报内容';
494:       _notify();
495:       return;
496:     }
497:     if (ttsBusy) return;
498:     ttsBusy = true;
499:     ttsMessage = '正在提交…';
500:     final operation = _operationGeneration;
501:     _notify();
502:     try {
503:       final commands = await gateway.sendTts(
504:         deviceIds: [device.id],
505:         text: text,
506:         eventId: eventId,
507:         idempotencyKey: _idempotencyKey('tts'),
508:       );
509:       if (_disposed || operation != _operationGeneration) return;
510:       final failed = commands.any(
511:         (item) => item.status == TtsCommandStatus.failed,
512:       );
513:       ttsMessage = failed ? '指令提交失败，请查看设备结果' : '指令已提交（不代表现场已听到）';
514:     } catch (error) {
515:       if (_disposed || operation != _operationGeneration) return;
516:       ttsMessage = _errorMessage(error, '指令提交失败');
517:     } finally {
518:       if (!_disposed && operation == _operationGeneration) {
519:         ttsBusy = false;
520:         _notify();
521:       }
522:     }
523:   }

532:   bool _applyServerUpdate(CallSession next) {
533:     final current = activeCall;
534:     if (current == null || current.id != next.id) return false;
535:     if (current.isTerminal && current.status != next.status) return false;
536:     if (current.version > 0 &&
537:         next.version > 0 &&
538:         next.version < current.version) {
539:       return false;
540:     }
541:     if (_statusRank(next.status) < _statusRank(current.status)) return false;
542:     activeCall = next;
543:     return true;
544:   }

546:   int _statusRank(WearCallStatus status) => switch (status) {
547:     WearCallStatus.unknown => -1,
548:     WearCallStatus.requesting => 0,
549:     WearCallStatus.offered => 1,
550:     WearCallStatus.connected => 2,
551:     WearCallStatus.ended ||
552:     WearCallStatus.failed ||
553:     WearCallStatus.timedOut => 3,
554:   };

556:   void _syncConnectionMessage() {
557:     final call = activeCall;
558:     if (call == null) return;
559:     if (call.demo) {
560:       statusMessage = call.statusLabel;
561:     } else if (call.status == WearCallStatus.connected) {
562:       if (rtcPhase == WearRtcPhase.failed) return;
563:       statusMessage = !_rtcActive || rtcPhase != WearRtcPhase.localJoined
564:           ? '服务器已确认，RTC 尚未就绪'
565:           : remotePresent
566:           ? '已接通'
567:           : '服务器已确认，等待设备加入…';
568:     } else {
569:       statusMessage = call.failReason.isNotEmpty
570:           ? '${call.statusLabel}：${call.failReason}'
571:           : call.statusLabel;
572:     }
573:   }

587:   void _completeJoin(int operation, {Object? error}) {
588:     if (_joinOutcomeOperation != operation) return;
589:     final outcome = _joinOutcome;
590:     _joinOutcome = null;
591:     _joinOutcomeOperation = null;
592:     if (outcome == null || outcome.isCompleted) return;
593:     if (error == null) {
594:       outcome.complete();
595:     } else {
596:       outcome.completeError(error);
597:     }
598:   }

600:   void _completeAnyJoin() {
601:     final outcome = _joinOutcome;
602:     _joinOutcome = null;
603:     _joinOutcomeOperation = null;
604:     if (outcome != null && !outcome.isCompleted) outcome.complete();
605:   }

608:     ++_operationGeneration;
609:     _sessionOperationPending = false;
610:     busy = false;
611:     _completeAnyJoin();

625:   String _errorMessage(Object error, String fallback) {
626:     final dynamic value = error;
627:     try {
628:       final message = value.message?.toString().trim();
629:       if (message != null && message.isNotEmpty) return '$fallback：$message';
630:     } catch (_) {}
631:     final text = error
632:         .toString()
633:         .replaceFirst(RegExp(r'^[A-Za-z]+(?:Exception|Error):\s*'), '')
