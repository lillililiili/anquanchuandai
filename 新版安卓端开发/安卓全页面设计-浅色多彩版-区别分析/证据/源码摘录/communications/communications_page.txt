117:   Future<void> _load() async {
118:     final session = _session;
119:     final gateway = _gateway;
120:     final controller = _controller;
121:     if (session == null || gateway == null || controller == null) return;
122:     final generation = ++_loadGeneration;
123:     final scope = session.scopeKey;
124:     final blocking = _people.isEmpty && _devices.isEmpty;
125:     setState(() {
126:       if (blocking) _loading = true;
127:       _loadError = null;
128:     });
129:     try {
130:       final people = await gateway.peopleOptions(
131:         name: _search.text.trim().isEmpty ? null : _search.text.trim(),
132:       );
133:       final devices = await gateway.listDevices(
134:         sn: _search.text.trim().isEmpty ? null : _search.text.trim(),
135:       );
136:       final myEquipment = await gateway.meEquipment();
137:       PersonOption? person;
138:       List<CommunicationDevice> equipment = const [];
139:       CommunicationDevice? selected;
140:       String eventType = '';
141:       String? resolvedDeviceId = _clean(widget.deviceId);
142:       final eventId = _clean(widget.eventId);
143: 
144:       if (eventId != null) {
145:         final event = await gateway.event(eventId);
146:         eventType = idOf(event['type']);
147:         resolvedDeviceId ??= _clean(idOf(event['deviceId']));
148:       }
149:       final personId = _clean(widget.personId);
150:       if (personId != null) {
151:         person =
152:             people.where((item) => item.id == personId).firstOrNull ??
153:             await gateway.person(personId);
154:         equipment = await gateway.equipmentForPerson(personId);
155:         selected = resolvedDeviceId == null
156:             ? equipment.firstOrNull
157:             : equipment
158:                   .where((item) => item.id == resolvedDeviceId)
159:                   .firstOrNull;
160:       }
161:       if (resolvedDeviceId != null && selected == null) {
162:         selected = await gateway.device(resolvedDeviceId);
163:         final assignedPersonId = selected.personId;
164:         if (person == null && assignedPersonId != null) {
165:           person = people
166:               .where((item) => item.id == assignedPersonId)
167:               .firstOrNull;
168:         }
169:       }
170:       if (selected != null &&
171:           equipment.every((item) => item.id != selected!.id)) {
172:         equipment = [...equipment, selected];
173:       }
174:       final history = await gateway.calls(
175:         eventId: eventId,
176:         deviceId: selected?.id,
177:       );
178:       if (!mounted ||
179:           generation != _loadGeneration ||
180:           scope != session.scopeKey) {
181:         return;
182:       }
183:       setState(() {
184:         _people = people;
185:         _devices = devices;
186:         _myEquipment = myEquipment;
187:         _equipment = equipment;
188:         _history = history;
189:         _eventType = eventType;
190:         _loading = false;
191:         if (selected != null && resolvedDeviceId != null) {
192:           _selectedKeys
193:             ..clear()
194:             ..add('d:${selected.id}');
195:         } else if (person != null) {
196:           _selectedKeys
197:             ..clear()
198:             ..add('p:${person.id}');
199:           _equipmentByPerson[person.id] = equipment;
200:         } else if (selected != null) {
201:           _selectedKeys
202:             ..clear()
203:             ..add('d:${selected.id}');
204:         }
205:       });
206:       controller.selectDevice(selected);
207:     } catch (error) {
208:       if (error is StaleSessionException) return;
209:       if (!mounted ||
210:           generation != _loadGeneration ||
211:           scope != session.scopeKey) {
212:         return;
213:       }
214:       setState(() {
215:         _loadError = error;
216:         _loading = false;
217:       });
218:     }
219:   }

221:   Future<void> _selectPerson(PersonOption? person) async {
222:     final session = _session;
223:     final gateway = _gateway;
224:     if (session == null || gateway == null || person == null) return;
225:     final generation = ++_loadGeneration;
226:     final scope = session.scopeKey;
227:     setState(() {
228:       _equipment = const [];
229:       _history = const [];
230:       _loadError = null;
231:     });
232:     _controller?.selectDevice(null);
233:     try {
234:       final equipment = await gateway.equipmentForPerson(person.id);
235:       final selected = equipment.firstOrNull;
236:       final history = await gateway.calls(deviceId: selected?.id);
237:       if (!mounted ||
238:           generation != _loadGeneration ||
239:           scope != session.scopeKey) {
240:         return;
241:       }
242:       setState(() {
243:         _equipment = equipment;
244:         _equipmentByPerson[person.id] = equipment;
245:         _history = history;
246:       });
247:       _controller?.selectDevice(selected);
248:     } catch (error) {
249:       if (error is StaleSessionException) return;
250:       if (!mounted ||
251:           generation != _loadGeneration ||
252:           scope != session.scopeKey) {
253:         return;
254:       }
255:       setState(() => _loadError = error);
256:     }
257:   }

259:   Future<void> _selectDevice(CommunicationDevice device) async {
260:     _controller?.selectDevice(device);
261:     final session = _session;
262:     final gateway = _gateway;
263:     if (session == null || gateway == null) return;
264:     final generation = ++_loadGeneration;
265:     final scope = session.scopeKey;
266:     try {
267:       final history = await gateway.calls(
268:         eventId: _clean(widget.eventId),
269:         deviceId: device.id,
270:       );
271:       if (!mounted ||
272:           generation != _loadGeneration ||
273:           scope != session.scopeKey) {
274:         return;
275:       }
276:       setState(() => _history = history);
277:     } catch (error) {
278:       if (error is StaleSessionException) return;
279:       if (mounted &&
280:           generation == _loadGeneration &&
281:           scope == session.scopeKey) {
282:         setState(() => _loadError = error);
283:       }
284:     }

1125:   Future<CommunicationDevice?> _deviceOf(_CommsContact contact) async {
1126:     if (contact.device != null) return contact.device;
1127:     final person = contact.person;
1128:     if (person == null) return null;
1129:     final cached = _equipmentByPerson[person.id];
1130:     if (cached != null) {
1131:       return cached.where((item) => item.supports('intercom')).firstOrNull ??
1132:           cached.firstOrNull;
1133:     }
1134:     final gateway = _gateway;
1135:     if (gateway == null) return null;
1136:     try {
1137:       final equipment = await gateway.equipmentForPerson(person.id);
1138:       if (mounted) {
1139:         setState(() => _equipmentByPerson[person.id] = equipment);
1140:       }
1141:       return equipment.where((item) => item.supports('intercom')).firstOrNull ??
1142:           equipment.firstOrNull;
1143:     } catch (_) {
1144:       return null;
1145:     }

1148:   Future<void> _startCallForSelection({
1149:     required bool video,
1150:     required bool fromHelmet,
1151:   }) async {
1152:     final controller = _controller;
1153:     if (controller == null) return;
1154:     if (fromHelmet && _boundHelmet == null) {
1155:       _snack('未绑定安全帽时，安全帽呼叫不可用');
1156:       return;
1157:     }
1158:     if (_selectedKeys.isEmpty) {
1159:       _snack('请先选择联系人');
1160:       return;
1161:     }
1162:     final resolved = <CommunicationDevice>[];
1163:     final labels = <String>[];
1164:     for (final contact in _allContacts) {
1165:       if (!_selectedKeys.contains(contact.key)) continue;
1166:       final device = await _deviceOf(contact);
1167:       if (device == null) continue;
1168:       final allowed = video
1169:           ? controller.policy.canStartVideo(device)
1170:           : controller.policy.canStartVoice(device);
1171:       if (!allowed) continue;
1172:       resolved.add(device);
1173:       labels.add(contact.title);
1174:     }
1175:     if (resolved.isEmpty) {
1176:       _snack(video ? '所选对象不支持视频通话' : '所选对象暂无可用通话装备');
