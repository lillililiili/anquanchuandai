1: typedef JsonMap = Map<String, dynamic>;
2: 
3: JsonMap jsonMap(Object? value) => value is Map
4:     ? value.map((key, value) => MapEntry(key.toString(), value))
5:     : <String, dynamic>{};
6: List<JsonMap> jsonList(Object? value) =>
7:     value is List ? value.whereType<Map>().map(jsonMap).toList() : <JsonMap>[];
8: String idOf(Object? value) => value == null ? '' : value.toString();
9: String textOf(Object? value, [String fallback = '—']) =>
10:     value == null || value.toString().trim().isEmpty
11:     ? fallback
12:     : value.toString();
13: int intOf(Object? value, [int fallback = 0]) => value is num
14:     ? value.toInt()
15:     : int.tryParse(value?.toString() ?? '') ?? fallback;
16: String formatTime(Object? value) {
17:   final date = DateTime.tryParse(value?.toString() ?? '')?.toLocal();
18:   if (date == null) return '时间未知';
19:   String two(int n) => n.toString().padLeft(2, '0');
20:   return '${date.year}-${two(date.month)}-${two(date.day)} ${two(date.hour)}:${two(date.minute)}:${two(date.second)}';
21: }
