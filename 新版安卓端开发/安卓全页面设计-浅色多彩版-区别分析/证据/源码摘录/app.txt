166:   Future<void> _openNotice(WearNotice notice) async {
167:     final user = _session.userId;
168:     if (_session.siteId != notice.siteId) {
169:       await _session.selectSite(notice.siteId);
170:     }
171:     final event = jsonMap(
172:       await _session.api.get('/api/v1/events/${notice.eventId}'),
173:     );
174:     if (_session.userId != user ||
175:         idOf(event['siteId']) != _session.siteId ||
176:         idOf(event['id']) != notice.eventId) {
177:       throw const StaleSessionException();
178:     }
179:     if (mounted) {
180:       _router.go('/events?eventId=${Uri.encodeComponent(notice.eventId)}');
181:       _session.requestRefresh();
182:     }

323: class _WearShellState extends State<WearShell> {
324:   WearSession? _session;
325:   int? _count;
326:   int _generation = 0;
327:   @override
328:   void didChangeDependencies() {
329:     super.didChangeDependencies();
330:     final session = WearScope.of(context);
331:     if (_session != session) {
332:       _session?.refreshTick.removeListener(_refresh);
333:       _session = session;
334:       session.refreshTick.addListener(_refresh);
335:       Future.microtask(_loadCount);
336:     }
337:   }
338: 
339:   void _refresh() {
340:     unawaited(_loadCount());
341:   }
342: 
343:   Future<void> _loadCount() async {
344:     if (_session == null || _session!.busy || _session!.siteId == null) return;
345:     final generation = ++_generation;
346:     try {
347:       final data = jsonMap(
348:         await _session!.api.get('/api/v1/events/inbox/count'),
349:       );
350:       if (mounted && generation == _generation) {
351:         setState(() => _count = intOf(data['count']));
352:       }
353:     } catch (_) {
354:       if (mounted && generation == _generation) setState(() => _count = null);
355:     }
356:   }
357: 
358:   @override
359:   void dispose() {
360:     _generation++;
361:     _session?.refreshTick.removeListener(_refresh);
362:     super.dispose();
363:   }
