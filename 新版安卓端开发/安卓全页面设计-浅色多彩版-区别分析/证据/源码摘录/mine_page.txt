41:   Future<void> _load() async {
42:     final request = ++_request;
43:     if (!mounted) return;
44:     setState(() {
45:       _loading = true;
46:       _error = null;
47:     });
48:     _detailTick.value++;
49:     try {
50:       final rows = jsonList(await _session!.api.get('/api/v1/duty/handovers'));
51:       if (mounted && request == _request) setState(() => _handovers = rows);
52:     } catch (e) {
53:       if (mounted && request == _request && e is! StaleSessionException) {
54:         setState(() => _error = e.toString());
55:       }
56:     } finally {
57:       if (mounted && request == _request) {
58:         setState(() => _loading = false);
59:         _detailTick.value++;
60:       }
61:     }
62:   }

101:         ScaffoldMessenger.of(

103:         ).showSnackBar(SnackBar(content: Text(e.toString())));
104:       }
105:     } finally {
106:       if (mounted) {
107:         setState(() => _confirming = null);
108:         _detailTick.value++;
109:       }
110:     }
111:   }
