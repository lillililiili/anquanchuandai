529:   Widget _memberStatus(JsonMap person, String type) {
530:     final items = equipment.where(
531:       (e) =>
532:           idOf(e['personId']) == idOf(person['personId']) &&
533:           e['typeCode'] == type,
534:     );
535:     final result = items.isEmpty ? null : items.first['result'];
536:     final color = result == 'ok'
537:         ? const Color(0xFF00BE95)
538:         : result == null
539:         ? muted
540:         : const Color(0xFFFFA000);
541:     final label = switch (type) {
542:       'helmet' => '帽',
543:       'belt' => '带',
544:       _ => '表',
545:     };
546:     // Equipment-check verifies readiness, not a live connectivity feed.
547:     final state = result == 'ok'
548:         ? '正常'
549:         : result == 'missing'
550:         ? '缺少'
551:         : result == null
552:         ? '未关联'
553:         : '待核验';
554:     return Row(
555:       mainAxisSize: MainAxisSize.min,
556:       children: [
557:         CustomPaint(
558:           size: const Size(20, 20),
559:           painter: _EquipmentGlyph(type, color),
560:         ),
561:         const SizedBox(width: 3),
562:         Text(
563:           '$label$state',
564:           style: TextStyle(fontSize: 11, color: result == 'ok' ? muted : color),
565:         ),
566:       ],
567:     );
568:   }
569: }

571: class _EquipmentGlyph extends CustomPainter {
572:   const _EquipmentGlyph(this.type, this.color);
573:   final String type;
574:   final Color color;
575:   @override
576:   void paint(Canvas canvas, Size size) {
577:     canvas.save();
578:     canvas.scale(size.width / 24, size.height / 24);
579:     final fill = Paint()..color = color;
580:     final stroke = Paint()
581:       ..color = color
582:       ..style = PaintingStyle.stroke
583:       ..strokeWidth = 2.5
584:       ..strokeJoin = StrokeJoin.round;
585:     if (type == 'helmet') {
586:       canvas.drawPath(
587:         Path()
588:           ..moveTo(3, 18)
589:           ..lineTo(3, 13)
590:           ..quadraticBezierTo(3, 5, 12, 4)
591:           ..quadraticBezierTo(21, 5, 21, 13)
592:           ..lineTo(21, 18)
593:           ..close(),
594:         fill,
595:       );
596:       canvas.drawRRect(
597:         RRect.fromRectAndRadius(
598:           const Rect.fromLTWH(10, 2, 4, 15),
599:           const Radius.circular(2),
600:         ),
601:         fill,
602:       );
603:       canvas.drawRRect(
604:         RRect.fromRectAndRadius(
605:           const Rect.fromLTWH(1, 17, 22, 5),
606:           const Radius.circular(2),
607:         ),
608:         fill,
609:       );
610:     } else if (type == 'belt') {
611:       canvas.drawRRect(
612:         RRect.fromRectAndRadius(
613:           const Rect.fromLTWH(5, 5, 14, 14),
614:           const Radius.circular(2),
615:         ),
616:         stroke,
617:       );
618:       canvas.drawRect(const Rect.fromLTWH(0, 8, 6, 8), fill);
619:       canvas.drawRect(const Rect.fromLTWH(18, 8, 6, 8), fill);
620:       canvas.drawLine(const Offset(13, 5), const Offset(13, 19), stroke);
621:     } else {
622:       canvas.drawRRect(
623:         RRect.fromRectAndRadius(
624:           const Rect.fromLTWH(8, 0, 8, 24),
625:           const Radius.circular(2),
626:         ),
627:         fill,
628:       );
629:       canvas.drawRRect(
630:         RRect.fromRectAndRadius(
631:           const Rect.fromLTWH(4, 5, 16, 14),
632:           const Radius.circular(3),
633:         ),
634:         fill,
635:       );
636:       canvas.drawLine(
637:         const Offset(12, 8),
638:         const Offset(12, 13),
639:         Paint()
640:           ..color = Colors.white
641:           ..strokeWidth = 2,
642:       );
643:       canvas.drawLine(
644:         const Offset(12, 13),
645:         const Offset(15, 13),
646:         Paint()
647:           ..color = Colors.white
648:           ..strokeWidth = 2,
649:       );
650:     }
651:     canvas.restore();
652:   }

654:   @override
655:   bool shouldRepaint(_EquipmentGlyph oldDelegate) =>
656:       type != oldDelegate.type || color != oldDelegate.color;
