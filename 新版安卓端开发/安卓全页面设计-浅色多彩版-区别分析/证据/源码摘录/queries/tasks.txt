46:   Future<void> _load({required int page}) async {
47:     final session = _session;
48:     if (session == null) return;
49:     final request = ++_request;
50:     final scopeKey = session.scopeKey;
51:     setState(() {
52:       _loading = true;
53:       _error = null;
54:     });
55:     try {
56:       final result = await session.api.page(
57:         _mine ? '/api/v1/work-tasks/mine' : '/api/v1/work-tasks',
58:         current: page,
59:         size: 20,
60:         query: {
61:           if (!_mine && _status != null) 'status': _status,
62:           if (!_mine && _workType != null) 'workType': _workType,
63:         },
64:       );
65:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
66:         return;
67:       }
68:       setState(() {
69:         _records = result.records;
70:         _current = result.current;
71:         _total = result.total;
72:         _hasMore = result.hasMore;
73:         _loading = false;
74:       });
75:     } catch (error) {
76:       if (error is StaleSessionException) return;
77:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
78:         return;
79:       }
80:       setState(() {
81:         _error = error;
82:         _loading = false;
83:       });
84:     }
85:   }

278:       _loading = true;

285:         session.api.get('/api/v1/work-tasks/${widget.id}/events'),
286:       ]);
287:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
288:         return;
289:       }
290:       setState(() {
291:         _task = jsonMap(results[0]);
292:         _equipment = jsonList(results[1]);
293:         _events = jsonList(results[2]);
294:         _loading = false;

297:       if (error is StaleSessionException) return;
298:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
299:         return;
