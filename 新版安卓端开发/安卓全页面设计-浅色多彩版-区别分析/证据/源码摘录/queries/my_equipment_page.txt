79:   Future<void> _load() async {
80:     final session = _session!;
81:     final scope = session.scopeKey;
82:     final request = ++_request;
83:     setState(() {
84:       _loading = true;
85:       _error = null;
86:     });
87:     try {
88:       final assignments = jsonList(
89:         await session.api.get('/api/v1/me/equipment'),
90:       );
91:       final rows = await Future.wait(
92:         assignments.map((row) async {
93:           final id = idOf(row['deviceId']);
94:           if (id.isEmpty) return row;
95:           try {
96:             final device = jsonMap(
97:               await session.api.get(
98:                 '/api/v1/devices/${Uri.encodeComponent(id)}',
99:               ),
100:             );
101:             return <String, dynamic>{
102:               ...row,
103:               'online': device['online'],
104:               'connectionQuality': device['connectionQuality'],
105:               'lastReportedAt': device['lastReportedAt'],
106:               'lastTelemetryAt': device['lastTelemetryAt'],
107:             };
108:           } on StaleSessionException {
109:             rethrow;
110:           } catch (_) {
111:             // Assignment remains usable when device telemetry is not readable.
112:             return <String, dynamic>{...row, 'telemetryUnavailable': true};
113:           }
114:         }),
115:       );
116:       if (!mounted || request != _request || scope != session.scopeKey) return;
117:       setState(() {
118:         _items = rows;
119:         _loading = false;
120:       });
121:     } catch (error) {
122:       if (!mounted ||
123:           error is StaleSessionException ||
124:           request != _request ||
125:           scope != session.scopeKey) {
126:         return;
127:       }
128:       setState(() {
129:         _error = error;
130:         _loading = false;
131:       });
132:     }
133:   }

678:   Future<List<JsonMap>> _fetchHistory(List<JsonMap> rows) async {
679:     final session = _session!;
680:     final scope = session.scopeKey;
681:     final ids = rows
682:         .map((r) => idOf(r['personId']))
683:         .where((id) => id.isNotEmpty)
684:         .toSet();
685:     if (ids.isEmpty) return [];
686:     final results = await Future.wait(
687:       ids.map(
688:         (id) async => jsonList(
689:           await session.api.get(
690:             '/api/v1/people/${Uri.encodeComponent(id)}/assignments',
691:           ),
692:         ),
693:       ),
694:     );
695:     if (scope != session.scopeKey) throw StaleSessionException();
696:     return results.expand((r) => r).toList();
697:   }
