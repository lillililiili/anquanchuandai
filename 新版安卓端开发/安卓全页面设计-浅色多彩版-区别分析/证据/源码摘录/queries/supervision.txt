7: List<JsonMap> aggregateLostSupervision(
8:   List<JsonMap> tasks,
9:   Map<String, List<JsonMap>> checksByTask,
10: ) {
11:   final people = <String, JsonMap>{};
12:   for (final task in tasks) {
13:     final taskId = idOf(task['id']);
14:     for (final check in checksByTask[taskId] ?? const <JsonMap>[]) {
15:       if (check['typeCode']?.toString() != 'helmet' ||
16:           check['result']?.toString() == 'ok') {
17:         continue;
18:       }
19:       final personId = idOf(check['personId']);
20:       if (personId.isEmpty) continue;
21:       final person = people.putIfAbsent(
22:         personId,
23:         () => <String, dynamic>{
24:           'personId': personId,
25:           'personName': textOf(check['personName'], '未知人员'),
26:           'tasks': <JsonMap>[],
27:         },
28:       );
29:       (person['tasks'] as List<JsonMap>).add({
30:         'taskId': taskId,
31:         'taskTitle': textOf(task['title']),
32:         'result': check['result'],
33:         'needsConfirm': check['needsConfirm'] == true,
34:       });
35:     }
36:   }
37:   return people.values.toList();
38: }

54:   @override
55:   void didChangeDependencies() {
56:     super.didChangeDependencies();
57:     final session = WearScope.of(context);
58:     if (!identical(session, _session)) {
59:       _session = session;
60:       _load();
61:     }
62:   }
63: 
64:   Future<void> _load() async {
65:     final session = _session;
66:     if (session == null) return;
67:     final request = ++_request;
68:     final scopeKey = session.scopeKey;
69:     setState(() {
70:       _loading = true;
71:       _error = null;
72:     });
73:     try {
74:       final tasks = <JsonMap>[];
75:       var current = 1;
76:       while (true) {
77:         final page = await session.api.page(
78:           '/api/v1/work-tasks',
79:           current: current,
80:           size: 100,
81:           query: const {'status': 'in_progress'},
82:         );
83:         if (!mounted || request != _request || scopeKey != session.scopeKey) {
84:           return;
85:         }
86:         tasks.addAll(page.records);
87:         if (!page.hasMore || page.records.isEmpty) break;
88:         current++;
89:       }
90: 
91:       final checks = <String, List<JsonMap>>{};
92:       const batchSize = 8;
93:       for (var offset = 0; offset < tasks.length; offset += batchSize) {
94:         final end = (offset + batchSize).clamp(0, tasks.length);
95:         final batch = tasks.sublist(offset, end);
96:         final responses = await Future.wait(
97:           batch.map(
98:             (task) => session.api.get(
99:               '/api/v1/work-tasks/${idOf(task['id'])}/equipment-check',
100:             ),
101:           ),
102:         );
103:         if (!mounted || request != _request || scopeKey != session.scopeKey) {
104:           return;
105:         }
106:         for (var index = 0; index < batch.length; index++) {
107:           checks[idOf(batch[index]['id'])] = jsonList(responses[index]);
108:         }
109:       }
110: 
111:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
112:         return;
113:       }
114:       setState(() {
115:         _people = aggregateLostSupervision(tasks, checks);
116:         _loading = false;
117:       });
118:     } catch (error) {
119:       if (error is StaleSessionException) return;
120:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
121:         return;
122:       }
123:       setState(() {
124:         _error = error;
125:         _loading = false;
126:       });
127:     }
128:   }
