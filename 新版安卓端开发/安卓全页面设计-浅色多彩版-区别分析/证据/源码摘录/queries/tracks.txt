147:   Future<void> _loadTracks() async {
148:     final session = _session;
149:     final person = _person;
150:     if (session == null || person == null) return;
151:     _stop();
152:     final request = ++_request;
153:     final scopeKey = session.scopeKey;
154:     setState(() {
155:       _preview = false;
156:       _loading = true;
157:       _error = null;
158:       _points = const [];
159:       _index = 0;
160:     });
161:     try {
162:       final loaded = <JsonMap>[];
163:       var pageNumber = 1;
164:       var expectedTotal = 0;
165:       while (loaded.length < 500) {
166:         final page = await session.api.page(
167:           '/api/v1/locations/people/${idOf(person['id'])}/tracks',
168:           current: pageNumber,
169:           size: 100,
170:           query: {
171:             'from': _from.toUtc().toIso8601String(),
172:             'to': _to.toUtc().toIso8601String(),
173:           },
174:         );
175:         if (!mounted || request != _request || scopeKey != session.scopeKey) {
176:           return;
177:         }
178:         expectedTotal = page.total.clamp(0, 500);
179:         loaded.addAll(page.records.take(500 - loaded.length));
180:         if (!page.hasMore ||
181:             loaded.length >= expectedTotal ||
182:             page.records.isEmpty) {
183:           break;
184:         }
185:         pageNumber++;
186:       }
187:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
188:         return;
189:       }
190:       setState(() {
191:         _points = loaded;
192:         _index = 0;
193:         _loading = false;
194:       });
195:     } catch (error) {
196:       if (error is StaleSessionException) return;
197:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
198:         return;
199:       }
200:       setState(() {
201:         _error = error;
202:         _loading = false;
203:       });
204:     }
205:   }

909:   Future<void> _pickDate({required bool from}) async {
910:     final initial = from ? _from : _to;
911:     final value = await showDatePicker(

913:       initialDate: initial,
914:       firstDate: DateTime(2020),
915:       lastDate: DateTime.now().add(const Duration(days: 1)),
916:     );

918:     final nextFrom = from ? _startOfDay(value) : _from;
919:     final nextTo = from ? _to : _endOfDay(value);
920:     if (nextTo.isBefore(nextFrom)) {
921:       ScaffoldMessenger.of(
922:         context,

927:       _from = nextFrom;
928:       _to = nextTo;
929:     });
930:     if (_preview) {
931:       _stop();
932:       setState(() => _index = 0);
933:     } else {
934:       _loadTracks();

1026: class _TrackPreviewPainter extends CustomPainter {
1027:   const _TrackPreviewPainter(this.index);
1028:   final int index;
1029:   static const route = [
1030:     Offset(.18, .78),
1031:     Offset(.24, .64),
1032:     Offset(.38, .64),
1033:     Offset(.52, .64),
1034:     Offset(.52, .48),
1035:     Offset(.52, .30),
1036:     Offset(.52, .22),
1037:     Offset(.65, .22),
1038:     Offset(.78, .22),
1039:   ];
1040:   @override
1041:   void paint(Canvas canvas, Size size) {
1042:     final points = route
1043:         .map((p) => Offset(p.dx * size.width, p.dy * size.height))
1044:         .toList();
1045:     final path = Path()..moveTo(points.first.dx, points.first.dy);
1046:     for (final point in points.skip(1)) {
1047:       path.lineTo(point.dx, point.dy);
1048:     }
1049:     canvas.drawPath(
1050:       path,
1051:       Paint()
1052:         ..color = Colors.white
1053:         ..strokeWidth = 6
1054:         ..style = PaintingStyle.stroke
1055:         ..strokeJoin = StrokeJoin.round,
1056:     );
1057:     canvas.drawPath(
1058:       path,
1059:       Paint()
1060:         ..color = const Color(0xFF008FFF)
1061:         ..strokeWidth = 3
1062:         ..style = PaintingStyle.stroke
1063:         ..strokeJoin = StrokeJoin.round,
1064:     );
1065:     final current = points[index.clamp(0, points.length - 1)];
1066:     canvas.drawCircle(current, 7, Paint()..color = Colors.white);
1067:     canvas.drawCircle(current, 5, Paint()..color = const Color(0xFF008FFF));
1068:     final label = TextPainter(
1069:       text: const TextSpan(
1070:         text: '陈建国 · 示例',
1071:         style: TextStyle(
1072:           color: Colors.white,
1073:           fontSize: 10,
1074:           fontWeight: FontWeight.w600,
1075:         ),
1076:       ),
1077:       textDirection: TextDirection.ltr,
1078:     )..layout();
1079:     final rect = Rect.fromLTWH(
1080:       (current.dx - label.width / 2 - 6).clamp(
1081:         4,
1082:         size.width - label.width - 16,
1083:       ),
1084:       current.dy - 30,
1085:       label.width + 12,
1086:       22,

1089:       RRect.fromRectAndRadius(rect, const Radius.circular(5)),
1090:       Paint()..color = const Color(0xFF008FFF),
