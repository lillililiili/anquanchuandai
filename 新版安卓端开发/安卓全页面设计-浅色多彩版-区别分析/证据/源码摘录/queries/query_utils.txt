93: int nearestPointIndex(List<JsonMap> points, DateTime target) {
94:   if (points.isEmpty) return -1;
95:   var best = 0;
96:   var bestDistance = _distance(points.first['occurredAt'], target);
97:   for (var index = 1; index < points.length; index++) {
98:     final distance = _distance(points[index]['occurredAt'], target);
99:     if (distance < bestDistance) {
100:       best = index;
101:       bestDistance = distance;
102:     }
103:   }
104:   return best;
105: }

107: Duration _distance(Object? raw, DateTime target) {
108:   final parsed = DateTime.tryParse(raw?.toString() ?? '');
109:   if (parsed == null) return const Duration(days: 365000);
110:   final difference = parsed.difference(target);
111:   return difference.isNegative ? -difference : difference;
112: }
