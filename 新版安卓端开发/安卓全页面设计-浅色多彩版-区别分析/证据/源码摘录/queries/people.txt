406:   @override

408:     final lat = double.tryParse(location?['lat']?.toString() ?? '');

417:     final quality = switch (location?['locationQuality']) {
418:       'ok' when hasPosition => '定位正常',
419:       'stale' => '定位已陈旧',
420:       _ => '定位状态未知',

423:       child: Column(
424:         crossAxisAlignment: CrossAxisAlignment.stretch,
425:         children: [
426:           Row(
427:             children: [
428:               const Icon(Icons.location_on_outlined, color: WearColors.brand),

432:                   '现场信息',
433:                   style: TextStyle(fontSize: 17, fontWeight: FontWeight.w700),
434:                 ),
435:               ),
436:               if (location?['demo'] == true)
437:                 const WearBadge(text: '示例数据', color: WearColors.warning),
438:             ],
439:           ),
440:           const SizedBox(height: 8),
441:           DetailField(
442:             label: '最近定位',
443:             value: hasPosition
