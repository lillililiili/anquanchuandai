36:   Future<void> _load({required int page}) async {
37:     final session = _session;
38:     if (session == null) return;
39:     final request = ++_request;
40:     final scopeKey = session.scopeKey;
41:     setState(() {
42:       _loading = true;
43:       _error = null;
44:     });
45:     try {
46:       final result = await session.api.page(
47:         '/api/v1/fences',
48:         current: page,
49:         size: 20,
50:       );
51:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
52:         return;
53:       }
54:       setState(() {
55:         _records = result.records;
56:         _current = result.current;
57:         _total = result.total;
58:         _hasMore = result.hasMore;
59:         _loading = false;
60:       });
61:     } catch (error) {
62:       if (error is StaleSessionException) return;
63:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
64:         return;
65:       }
66:       setState(() {
67:         _error = error;
68:         _loading = false;
69:       });
70:     }

326:   List<LatLng> _polygon(Object? value) {
327:     return jsonList(value)
328:         .map((point) {
329:           final lat = double.tryParse(point['lat']?.toString() ?? '');
330:           final lng = double.tryParse(point['lng']?.toString() ?? '');
331:           return lat == null || lng == null ? null : LatLng(lat, lng);
332:         })
333:         .whereType<LatLng>()
334:         .toList();
335:   }

337:   String _timeWindow(JsonMap fence) {
338:     if (fence['timeStart'] == null && fence['timeEnd'] == null) return '全天';
339:     return '${textOf(fence['timeStart'], '00:00')}–${textOf(fence['timeEnd'], '24:00')}';
340:   }

342:   String _direction(JsonMap fence) {
343:     final values = <String>[];
344:     if (fence['enterEnabled'] == true) values.add('进入');
345:     if (fence['leaveEnabled'] == true) values.add('离开');
346:     return values.isEmpty ? '未启用' : values.join('、');
