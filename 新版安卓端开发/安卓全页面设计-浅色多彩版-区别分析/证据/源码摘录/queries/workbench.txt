49:   Future<void> _load({bool silent = false}) async {
50:     final session = _session;
51:     if (session == null) return;
52:     final request = ++_request;
53:     final scopeKey = session.scopeKey;
54:     if (!silent || _summary == null) {
55:       setState(() {
56:         _loading = true;
57:         _error = null;
58:       });
59:     }
60:     try {
61:       final result = jsonMap(await session.api.get('/api/v1/duty/summary'));
62:       List<JsonMap> equipment = const [];
63:       var equipmentUnavailable = false;
64:       try {
65:         equipment = jsonList(await session.api.get('/api/v1/me/equipment'));
66:       } catch (_) {
67:         equipmentUnavailable = true;
68:       }
69:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
70:         return;
71:       }
72:       setState(() {
73:         _summary = result;
74:         _equipment = equipment;
75:         _equipmentUnavailable = equipmentUnavailable;
76:         _loading = false;
77:         _error = null;
78:       });
79:     } catch (error) {
80:       if (error is StaleSessionException) return;
81:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
82:         return;
83:       }
84:       setState(() {
85:         _error = error;
86:         _loading = false;
87:       });
88:     }
89:   }
