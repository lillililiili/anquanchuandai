44:   Future<void> _load({required int page}) async {
45:     final session = _session;
46:     if (session == null) return;
47:     final request = ++_request;
48:     final scopeKey = session.scopeKey;
49:     setState(() {
50:       _loading = true;
51:       _error = null;
52:     });
53:     try {
54:       final result = await session.api.page(
55:         '/api/v1/devices',
56:         current: page,
57:         size: 20,
58:         query: {
59:           if (_sn.text.trim().isNotEmpty) 'sn': _sn.text.trim(),
60:           if (_typeCode != null) 'typeCode': _typeCode,
61:           if (_assetStatus != null) 'assetStatus': _assetStatus,
62:         },
63:       );
64:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
65:         return;
66:       }
67:       setState(() {
68:         _records = result.records;
69:         _current = result.current;
70:         _total = result.total;
71:         _hasMore = result.hasMore;
72:         _loading = false;
73:       });
74:     } catch (error) {
75:       if (error is StaleSessionException) return;
76:       if (!mounted || request != _request || scopeKey != session.scopeKey) {
77:         return;
78:       }
79:       setState(() {
80:         _error = error;
81:         _loading = false;
82:       });
83:     }
84:   }
