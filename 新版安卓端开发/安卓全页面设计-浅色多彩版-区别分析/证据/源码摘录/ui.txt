54: class _RollingMarkPainter extends CustomPainter {
55:   const _RollingMarkPainter();
56: 
57:   @override
58:   void paint(Canvas canvas, Size size) {
59:     final stroke = size.width * 0.2;
60:     final ring = Paint()
61:       ..color = WearColors.brand
62:       ..style = PaintingStyle.stroke
63:       ..strokeWidth = stroke;
64:     final fill = Paint()..color = WearColors.brand;
65:     final c = Offset(size.width / 2, size.height / 2);
66:     final r = size.width / 2 - stroke / 2;
67:     canvas.drawCircle(c, r, ring);
68:     canvas.drawCircle(c, r * 0.36, fill);
69:   }
70: 
71:   @override
72:   bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
73: }
